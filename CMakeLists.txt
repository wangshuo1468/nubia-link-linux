# All automated tests use local fixtures; no phone operations are registered.
function(nubia_test target source library)
  add_executable(${target} tests/${source}.cpp)
  target_link_libraries(${target} PRIVATE ${library} Qt6::Test)
  if(ARGC GREATER 3)
    add_test(NAME ${ARGV3} COMMAND ${target})
  endif()
endfunction()
